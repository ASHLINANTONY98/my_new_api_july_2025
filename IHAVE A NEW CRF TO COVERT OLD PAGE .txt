        function handleFileUpload(fileInput, photoNameField, base64Field) {

        // Check if a file is selected
        if (fileInput.files.length === 0) {
        invoke_alert('Please select a file to upload', "warning");
        return;
        }

        const file = fileInput.files[0];
        const fileType = file['type'];
        const validImageTypes = ['image/jpeg', 'image/png', 'image/jpg'];

        // Check file type
        if (!validImageTypes.includes(fileType)) {
        invoke_alert('Please upload file in JPEG/JPG/PNG file format', "warning");
        fileInput.value = '';
        return;
        }

        // Check file size
        const fileSize = (file.size / 1024 / 1024).toFixed(2);
        if (fileSize > 3) {
        invoke_alert("Attached file size exceeds 3 MB, Please attach file less than or equal to 3 MB", "warning");
        fileInput.value = '';
        return;
        }

        // Create a URL for the file
        const fileURL = URL.createObjectURL(file);


        // Get the file as a base64 string
        fetch(fileURL)
        .then(response => response.blob())
        .then(blob => {
        const reader = new FileReader();
        reader.onload = function (event) {
        const base64Url = event.target.result; // Get the Base64 string
        base64Field.value = base64Url;
        const fileName = file.name;
        photoNameField.value = fileName; // Set the file name
        invoke_alert('File Uploaded successfully', "success");
        };
        reader.readAsDataURL(blob);
        })
        .catch(error => {
        invoke_alert('Error reading file', "error");
        });

        // Clean up
        URL.revokeObjectURL(fileURL);
        }
