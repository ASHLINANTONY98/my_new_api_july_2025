        // ---- API keys: must match what is registered in your HRMSModuleAPI ----
        // empInfoKey -> proc flag 1 (empcode, name, designation)
        // The save call goes through AssetandLiabilityController.SaveEmployeeFamily (flag 2).
        var AL = {
            module: "ASSET_AND_LIABILITY",
            nextPage: "AssetandLiability/Movable_forms"
        };

        var MAX_DEPENDANTS = 3;

        // ------------------------------------------------------------
        //  URL helpers (same localhost / root-folder logic as before)
        // ------------------------------------------------------------
        function appUrl(path) {
            var root = '@ViewData["root"]';

            if (document.location.hostname == 'localhost' || !root) {
                return "/" + path;
            }
            return "/" + root + "/" + path;
        }

        function GetLink() {
            return appUrl("AssetandLiability/getAPIData");
        }

        function goToDashboard() {
            window.location.href = appUrl("Home/Dashboard");
        }

        // ------------------------------------------------------------
        //  Small helpers
        // ------------------------------------------------------------
        function warn(msg) {
            Swal.fire({ icon: "warning", text: msg });
        }

        function firstRow(response) {
            try {
                var data = (typeof response === "string") ? JSON.parse(response) : response;
                if (Array.isArray(data)) return data[0] || null;
                if (data && Array.isArray(data.RESP)) return data.RESP[0] || null;
                return data || null;
            } catch (e) {
                return null;
            }
        }

        function pick(row, name) {
            if (!row || typeof row !== "object") return "";
            for (var k in row) {
                if (k.toUpperCase() === name) return row[k] == null ? "" : String(row[k]);
            }
            return "";
        }

        function setIf(selector, value) {
            if (value) $(selector).val(value);
        }

        // "111" = inserted, "222" = already exists (same codes the old createnew() returned)
        function resultCode(response) {
            var data = response;
            try { data = (typeof response === "string") ? JSON.parse(response) : response; } catch (e) { data = response; }
            if (Array.isArray(data)) data = data[0];
            if (data && typeof data === "object") data = Object.values(data)[0];
            var m = String(data == null ? "" : data).split("*")[0].match(/\b(111|222)\b/);
            return m ? m[1] : "";
        }

        // ------------------------------------------------------------
        //  Page load : employee details (old emp_load)
        // ------------------------------------------------------------
        function loadEmployee() {
            var empCode = $("#txt_empcode").val();

            if (!empCode) {
                Swal.fire({ icon: "error", title: "Session expired", text: "Please log in again." });
                return;
            }

            var indata = AL.module + "§" + "1" + "‡" + empCode;

            $.ajax({
                type: "GET",
                url: GetLink(),
                data: { datas: indata },
                success: function (response) {
                    var row = firstRow(response);
                    if (!row) return;
                    setIf("#txt_empcode", pick(row, "EMP_CODE"));
                    setIf("#txt_empnme", pick(row, "EMP_NAME"));
                    setIf("#txt_desi", pick(row, "DESIGNATION"));
                },
                error: function () {
                    warn("Could not load employee details.");
                }
            });
        }

        // letters and space only, upper case (also blocks pasted text)
        $(document).on("input", ".alpha-only", function () {
            this.value = this.value.replace(/[^A-Za-z ]/g, "").toUpperCase();
        });

        // ------------------------------------------------------------
        //  Spouse
        // ------------------------------------------------------------
        function toggleSpouseDetails() {
            var hasSpouse = $("#rdo_spo3").is(":checked");
            $("#spo_div1").toggleClass("d-none", !hasSpouse);

            if (!hasSpouse) clearTextField();
        }

        function clearTextField() {
            $("#txt_spousename, #txt_spouseposition").val("");
            $('input[name="radio_spo2"]').prop("checked", false);
        }

        // ------------------------------------------------------------
        //  Dependants
        // ------------------------------------------------------------
        function showDependentDetails() {
            var selected = $("#ddl_local2").val();

            // clear the fields every time a selection is made
            $("#txt_depname, #txt_deppositn, #txt_relation").val("");
            $("#dependentDetails").toggleClass("d-none", selected === "0");
        }

        function clearInputs() {
            $("#txt_depname, #txt_deppositn, #txt_relation").val("");
            $("#ddl_local2").val("0");
            $('input[name="radio_spo3"]').prop("checked", false);
            $("#dependentDetails").addClass("d-none");
        }

        function refreshDependentTable() {
            $("#tbl_ba").toggleClass("d-none", $("#tbl_viewba tbody tr").length === 0);
        }

        function add_rect() {
            var dependant = $("#ddl_local2").val();
            var label = $("#ddl_local2 option:selected").text();
            var name = $.trim($("#txt_depname").val());
            var position = $.trim($("#txt_deppositn").val());
            var relation = $.trim($("#txt_relation").val());
            var filed = $('input[name="radio_spo3"]:checked').val();   // YES / NO

            if (dependant === "0" || !name || !position || !relation || !filed) {
                warn("Please enter all the fields..!");
                return false;
            }

            var already = $("#tbl_viewba tbody tr").filter(function () {
                return $(this).children("td").eq(0).text() === label;
            }).length;

            if (already > 0) {
                warn("Already added this dependant !!!");
                clearInputs();
                return false;
            }

            if ($("#tbl_viewba tbody tr").length >= MAX_DEPENDANTS) {
                warn("Add only three dependant details.");
                clearInputs();
                return false;
            }

            var $del = $("<i>", { "class": "bi bi-trash text-danger", role: "button", title: "Remove" })
            .on("click", function () { dlt_s1(this); });

            var $tr = $("<tr>");
            $tr.append(
                $("<td>").text(label),
                $("<td>").text(name),
                $("<td>").text(position),
                $("<td>").text(relation),
                $("<td>").text(filed),
                $("<td>").append($del)
            );

            $("#tbl_viewba tbody").append($tr);
            refreshDependentTable();
            clearInputs();
        }

        function dlt_s1(el) {
            $(el).closest("tr").remove();
            refreshDependentTable();
        }

        // ------------------------------------------------------------
        //  CONFIRM (old insert)
        // ------------------------------------------------------------
        function insert() {
            var emp_code = $("#txt_empcode").val();
            var emp_name = $("#txt_empnme").val();
            var posi = $("#txt_desi").val();

            // ---- spouse ----
            var spouseName, spousePos, spouseFiled;

            if ($("#rdo_spo3").is(":checked")) {
                spouseName = $.trim($("#txt_spousename").val());
                spousePos = $.trim($("#txt_spouseposition").val());

                if (spouseName === "") { warn("Please Add Spouse Name...!!"); return false; }
                if (spousePos === "") { warn("Please Add Spouse position...!!"); return false; }

                spouseFiled = $('input[name="radio_spo2"]:checked').val();
                if (spouseFiled === undefined) { warn("Please select whether filled by him/her...!!"); return false; }
            }
            else if ($("#rdo_spo4").is(":checked")) {
                spouseName = "nill";
                spousePos = "nill";
                spouseFiled = 0;
            }
            else {
                warn("Please select spouse ...!!");
                return false;
            }

            // ---- employee: filed by him/her ----
            var empFiled = $('input[name="radio_spo"]:checked').val();
            if (empFiled === undefined) { warn("Please select whether filled by him/her...!!"); return false; }

            // ---- dependants ----
            var dependents = "";
            $("#tbl_viewba tbody tr").each(function () {
                var c = $(this).children("td");
                var whrk = $.trim(c.eq(4).text()) === "YES" ? 1 : 0;

                dependents += [c.eq(0).text(), c.eq(1).text(), c.eq(2).text(), c.eq(3).text(), whrk].join("*") + "!";
            });

            if (dependents === "") { warn("Please Add dependent details ...!!"); return false; }

            // same layout as the old page: head # dependants
            var head = [emp_code, emp_name, posi, empFiled, spouseName, spousePos, spouseFiled].join("~");
            var payload = head + "#" + dependents;

            $("#btnConfirm").prop("disabled", true);

            $.ajax({
                type: "POST",
                url: appUrl("AssetandLiability/SaveEmployeeFamily"),
                headers: { "RequestVerificationToken": $('input[name="__RequestVerificationToken"]').val() },
                data: { datas: payload },
                success: function (response) {
                    var code = resultCode(response);

                    if (code === "111") {
                        Swal.fire({ icon: "success", title: "Insert Successfully" }).then(function () {
                            window.location.href = appUrl(AL.nextPage);
                        });
                    }
                    else if (code === "222") {
                        Swal.fire({ icon: "info", title: "Already Exits!!!!" }).then(function () {
                            window.location.reload();
                        });
                    }
                    else {
                    $("#btnConfirm").prop("disabled", false);
                    Swal.fire({ icon: "error", title: "Not saved", text: "Please try again." });
                    }
                },
                error: function () {
                    $("#btnConfirm").prop("disabled", false);
                    Swal.fire({ icon: "error", title: "Failed", text: "Could not save. Please try again." });
                }
            });
        }

        // ------------------------------------------------------------
        //  Start
        // ------------------------------------------------------------
        $(function () {
            $("nav[aria-label=breadcrumb] a[data-page]").each(function () {
                $(this).attr("href", appUrl($(this).data("page")));
            });

            loadEmployee();
        });

    </script>

        function parseJson(response) {
            if (typeof response !== "string") return response;
            try { return JSON.parse(response); } catch (e) { }

        function addRow(key) {
            var b = BMAP[key];
            var vals = readValues(b);

            for (var i = 0; i < b.fields.length; i++) {
                var f = b.fields[i];
                if (isShown(b, f) && f.req !== false && vals[f.id] === "") {
                    warn("Please enter all the fields..!");
                    return false;
                }
            }

            var cells = b.fields.map(function (f) {
                return (isShown(b, f) && vals[f.id] !== "") ? vals[f.id] : "-";
            });

            // an identical row is a duplicate (the same type can be added again with different details)
            var signature = cells.join("|");
            var duplicate = $("#tbl_" + key + " tbody tr").filter(function () {
                return $(this).data("sig") === signature;
            }).length;

            if (duplicate > 0) {
                warn("Already added this entry !!!");
                return false;
            }

            var $tr = $("<tr>").data("sig", signature);
            cells.forEach(function (c) { $tr.append($("<td>").text(c)); });
            $tr.append($("<td>").append(
                $("<i>", { "class": "bi bi-trash text-danger btn-del", role: "button", title: "Remove" })
            ));

            $("#tbl_" + key + " tbody").append($tr);
            $("#tbl_" + key).removeClass("d-none");
            resetFields(b);
        }

            return s;
        }

        function insert() {

            var dependant = $("#ddl_emp").val();
            var seq = currentSeq();

            if (dependant === "-1" || dependant === null || !seq) { warn("please select dependant..!"); return false; }

            var cash = $.trim($("#txt_cash").val());
            if (cash === "") { warn("please Add Cash And BankBalance..!"); return false; }

            var balRmk = $.trim($("#txt_remark").val());
            if (balRmk === "") { warn("please Add Cash And BankBalance Remark..!"); return false; }

            var loanWrk;
            if ($("#rdo_spo1").is(":checked")) loanWrk = 1;
            else if ($("#rdo_spo2").is(":checked")) loanWrk = 0;
            else { warn("please select personalLoan ..!"); return false; }

            var loanRmk = $.trim($("#txt_rmrk").val());
            if (loanRmk === "") { warn("please Add personalLoan Remark..!"); return false; }

            // Yes / No on every block, and at least one row when the answer is Yes
            var segs = {};
            for (var i = 0; i < BLOCKS.length; i++) {
                var b = BLOCKS[i];
                var answer = blockAnswer(b);

                if (answer === undefined) { warn(b.msg); return false; }

                if (answer === "1") {
                    if ($("#tbl_" + b.key + " tbody tr").length === 0) {
                        warn("No rows have been inserted. Please add at least one row before proceeding.");
                        return false;
                    }
                    segs[b.seg] = blockRows(b);
                }
                else {
                    segs[b.seg] = "";
                }
            }

            // same header as the old page; the server replaces the last item with the logged-in user
            var head = [cash, balRmk, loanWrk, loanRmk, dependant, seq, empCode()].join("~");

            var payload = head;
            for (var s = 2; s <= 8; s++) payload += "#" + (segs[s] || "");

            $("#btnConfirm").prop("disabled", true);

            $.ajax({
                type: "POST",
                url: appUrl("AssetandLiability/SaveMovable"),
                headers: { "RequestVerificationToken": $('input[name="__RequestVerificationToken"]').val() },
                data: { datas: payload },
                success: function (response) {
                    var code = resultCode(response);

                    if (code === "111") {
                        Swal.fire({ icon: "success", title: "Insert Successfully" }).then(function () {
                            // last pending person -> next page, otherwise the next person
                            if (remaining === 1) window.location.href = appUrl(AL.nextPage);
                            else window.location.reload();
                        });
                    }
                    else if (code === "222") {
                        Swal.fire({ icon: "info", title: "Already Exits!!!!" }).then(function () {
                            window.location.reload();
                        });
                    }
                    else {
                        $("#btnConfirm").prop("disabled", false);
                        Swal.fire({ icon: "error", title: "Not saved", text: "Please try again." });
                    }
                },
                error: function () {
                    $("#btnConfirm").prop("disabled", false);
                    Swal.fire({ icon: "error", title: "Failed", text: "Could not save. Please try again." });
