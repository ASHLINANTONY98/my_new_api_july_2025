    select dependant,
           cash_and_bankbalance,

    (
        select sequence

        where seq_id = v_spliter_a(1)

  elsif p_flag = 34 then
    open p_as_outresult for
      select dependant, investment, amount, remark, issuer, units_qty
        from TBL_HRM_ASSET_INVESTMENT

  elsif p_flag = 35 then
    open p_as_outresult for
      select dependant, other_assets, value, remark, source_of_funds

  elsif p_flag = 36 then

  elsif p_flag = 37 then
    open p_as_outresult for

    open p_as_outresult for

      v_spliter_b1 := splitstr(v_spliter_b(2), '!');

          insert into TBL_HRM_ASSET_OTHER_ACCOUNTS (SEQ_ID, DEPENDANT, INSTITUTION, HOLDING_TYPE, APPROX_VALUE)
          values (seq1, v_spliter_c(3), v_spliter_c(1), v_spliter_c(2), to_number(v_spliter_c(4)));
        end if;
      end loop;
    end if;

    commit;
    open p_as_outresult for select '111' from dual;
    return;

  ----------------------------------------------------------------------
 
  end if;
exception

    values
      (v_ErrID,
       'Proc_Hrm_Asset_And_Liability_mvc_new',
       v_ErrDesc,
       v_ErrDtl,
       sysdate);
    commit;
end Proc_Hrm_Asset_And_Liability_mvc_new;

        function appUrl(path) {
            var root = '@ViewData["root"]';

            if (document.location.hostname == 'localhost' || !root) {
                return "/" + path;
            }
            return "/" + root + "/" + path;
        }

        function GetLink() {
            return appUrl("AssetandLiability/getAPIData");

        function warn(msg) {

        // getAPIData removes {"RESP": and can leave a closing } behind
        function parseJson(response) {
            if (typeof response !== "string") return response;
            try { return JSON.parse(response); } catch (e) { }
            try { return JSON.parse(response.replace(/\}\s*$/, "")); } catch (e) { }
            return null;
        }

        function rowsOf(response) {
            var data = parseJson(response);
            if (Array.isArray(data)) return data;
            if (data && Array.isArray(data.RESP)) return data.RESP;
            return data ? [data] : [];
        }

        function pick(row, name) {
            if (!row || typeof row !== "object") return "";
            for (var k in row) {
                if (k.toUpperCase() === name) return row[k] == null ? "" : String(row[k]);
            }
            return "";
        }

        function firstValue(response) {
            var row = rowsOf(response)[0];
            if (row && typeof row === "object") return Object.values(row)[0];
            return row;
        }

        // "111" = saved, "222" = already saved
        function resultCode(response) {
            var data = parseJson(response);
            if (data === null) data = response;
            if (Array.isArray(data)) data = data[0];
            if (data && typeof data === "object") data = Object.values(data)[0];
            var m = String(data == null ? "" : data).split("*")[0].match(/\b(111|222)\b/);
            return m ? m[1] : "";
        }

        function insert() {
            var seq = $("#ddl_emp").val();
            if (seq === "-1" || seq === null) { warn("Please select a dependent."); return; }

            var hasDebt = $("#rdo_debt_yes").is(":checked");
            var noDebt = $("#rdo_debt_no").is(":checked");

                warn("No rows have been inserted. Please add at least one row before proceeding.");
                return;
            }

            var dependent = $.trim($("#ddl_emp option:selected").text().split("-")[0]);
            var action, payload;

            if (hasDebt) {
                var rows = "";
                $("#deptTable tbody tr").each(function () { rows += $(this).data("vals").join("*") + "!"; });
                action = "SaveDebt";
                payload = dependent + "#" + seq + "#" + rows;

                action = "SaveNoDebt";
                payload = dependent + "~" + seq;

                var code = resultCode(response);

                }
            }, function () {
                $("#btnConfirm").prop("disabled", false);

            });
        }

        public string getAPIData(string datas)
            {
            string ApiPath = "hrmsmoduleapi/api/HRMSModuleAPI/GetDataHRMS/";
            ApiPath = baseurl + ApiPath;
            var token = HttpContext.Session.GetString("TokenValAPI");


            string[] DataArray = datas.Split('§');

            ViewData["baseurl"] = baseurl;
            ViewData["root"] = rootfolder;
            ViewData["HeadName"] = datas;
            string flag = DataArray[0];
            string indata = DataArray[1];
            try
            {
                var resData = _Grepo.GetInternalPageData(indata, flag, ApiPath, token);
                resData = resData.Replace(@"{""RESP"":", @"");

                return resData;
            }
            catch (Exception ex)
            {
                return ex.Message;
            }
        }

        public async Task<dynamic> postAPIData(string datas)
        {

            string ApiPath = "hrmsmoduleapi/api/HRMSModuleAPI/PostDataHRMS";
            ApiPath = baseurl + ApiPath;
            var token = HttpContext.Session.GetString("TokenValAPI");
            string[] DataArray = datas.Split('^');
            string flag = DataArray[0];
            string indata = DataArray[1];
            try
            {
                var response = await _Prepo.PostInternalPageData(indata, flag, ApiPath, token);

                return response.ToString();
            }
            catch (Exception ex)
            {
                if (ex.Message.Contains("Token has expired"))
                {
                    return null;
                }
                return null;
            }

        }

        {

                if (string.IsNullOrEmpty(empCode)) return "ERR";

                // header sequence of the latest declaration (old flag 29)
                DataTable dtSeq = ApiTable(29, empCode);
                if (dtSeq.Rows.Count == 0) return "ERR";
                string seqId = Convert.ToString(dtSeq.Rows[0][0]);

                var data = new AssetLiabilityPdfData
                {
                    Header = ApiTable(39, seqId),
                    Family = ApiTable(28, empCode),

                    Investments = ApiTable(34, seqId),
                    OtherAssets = ApiTable(35, seqId),

                    Jewellery = ApiTable(32, seqId),
                    Bank = ApiTable(40, seqId),
                    Insurance = ApiTable(41, seqId),
                    Vda = ApiTable(42, seqId),
                    RecentAcq = ApiTable(43, seqId),
                    IncomeAnnual = ApiTable(44, seqId),
                    IncomeOther = ApiTable(45, seqId),
                    OtherAccounts = ApiTable(46, seqId)
                };

                byte[] pdf;
                using (var ms = new MemoryStream())
                {
                    new AssetLiabilityPdfBuilder(ALWatermarkPath, Environment.GetFolderPath(Environment.SpecialFolder.Fonts)).Build(ms, data);
                    pdf = ms.ToArray();
                }

                string fileName = empCode + "_ASSETLIABILITY_" + DateTime.Now.ToString("yyyyMMddHHmmss") + ".pdf";

                await UploadAssetLiabilityPdf(seqId, empCode, fileName, pdf);

                return fileName;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Asset & Liability PDF generation failed");
                return "ERR";
            }

        private static DataTable ToDataTable(string json)
        {
            if (string.IsNullOrWhiteSpace(json)) return new DataTable();
            json = json.Trim();

            // getAPIData strips {"RESP": and can leave a closing } behind
            try { return JsonConvert.DeserializeObject<DataTable>(json) ?? new DataTable(); } catch { }
            try
            {
                string trimmed = System.Text.RegularExpressions.Regex.Replace(json, @"\}\s*$", "");
                return JsonConvert.DeserializeObject<DataTable>(trimmed) ?? new DataTable();
            }
            catch { }

            return new DataTable();
        }

    function UploadDocument(empCode, docId, file, description) {

        if (!file)
            return;

        var reader = new FileReader();

        reader.onload = function (e) {

            var base64 = e.target.result.split(',')[1];

            var ext = file.name.split('.').pop();

            // STEP 1 : Execute CLEANERDOCUPLOAD procedure
            var uploadQuery =
                GetCleanerUploadQuery(
                    empCode,
                    docId,
                    description,
                    ext,
                    file.name
                );

            console.log("Upload Query => ", uploadQuery);

            if (!uploadQuery || uploadQuery == "") {

                Swal.fire({
                    icon: "error",
                    title: "Error",
                    text: "Upload query not generated"
                });

                return;
            }

            // STEP 2 : Upload document using returned SQL
            var indata = uploadQuery + "~" + base64;

            var _link =
                document.location.hostname == "localhost"
                    ? "/Cleaner_Joining/DocumentUpload"
                    : "/" + '@ViewData["root"]' +
                      "/Cleaner_Joining/DocumentUpload";

            $.ajax({
                type: "POST",
                url: _link,
                data: {
                    datas: indata
                },
                async: false,
                success: function (response) {

                    console.log(description + " uploaded");
                    console.log(response);

                },
                error: function (xhr) {

                    console.log(xhr.responseText);

                    Swal.fire({
                        icon: "error",
                        title: "Upload Failed",
                        text: description + " upload failed"
                    });
                }
            });
        };

        reader.readAsDataURL(file);
    }

        public string getAPIData(string datas)
        {
            string ApiPath = "hrmsmoduleapi/api/HRMSModuleAPI/GetDataHRMS/";
            ApiPath = baseurl + ApiPath;
            var token = HttpContext.Session.GetString("TokenValAPI");


            string[] DataArray = datas.Split('^');

            ViewData["baseurl"] = baseurl;
            ViewData["root"] = rootfolder;
            ViewData["HeadName"] = datas;
            string flag = DataArray[0];
            string indata = DataArray[1];
            try
            {
                var resData = _Grepo.GetInternalPageData(indata, flag, ApiPath, token);
                resData = resData.Replace(@"{""RESP"":", @"");

                return resData;
            }
            catch (Exception ex)
            {
                if (ex.Message.Contains("Token has expired"))
                {
                    return null;
                }
                return null;
            }
        }
        public async Task<dynamic> postAPIData(string datas)

            }

        }

        public IActionResult ExportCleanerReport(string indata)
        {
            string ApiPath = "hrmsmoduleapi/api/HRMSModuleAPI/GetDataHRMS/";
            ApiPath = baseurl + ApiPath;

            var token = HttpContext.Session.GetString("TokenValAPI");

            string[] arr = indata.Split('*');

            string reportData = arr[1];    // 21~branch~emp~status~from~to

            string reportType = reportData.Split('~')[0];

            var response = _Grepo.GetInternalPageData(
                indata,
                "CLEANER_JOINING",
                ApiPath,
                token);

            response = response.Replace(@"{""RESP"":", "");

            var data = JsonConvert.DeserializeObject<List<JObject>>(response);


            if (data == null || data.Count == 0)
            {
                return Content("No Data Found");
            }

            IWorkbook workbook = new XSSFWorkbook();

            ISheet sheet = workbook.CreateSheet(
                reportType == "21"
                ? "Punching Report"
                : "Master Report"
            );

            // Header Style
            ICellStyle headerStyle = workbook.CreateCellStyle();

            IFont font = workbook.CreateFont();
            font.IsBold = true;

            headerStyle.SetFont(font);

            // Header Row
            IRow headerRow = sheet.CreateRow(0);

            if (reportType == "21")
            {
                headerRow.CreateCell(0).SetCellValue("Employee Code");
                headerRow.CreateCell(1).SetCellValue("Cleaner Name");
                headerRow.CreateCell(2).SetCellValue("Branch ID");
                headerRow.CreateCell(3).SetCellValue("Branch Name");
                headerRow.CreateCell(4).SetCellValue("Punch Date");
                headerRow.CreateCell(5).SetCellValue("Punch In");
                headerRow.CreateCell(6).SetCellValue("Punch Out");
                headerRow.CreateCell(7).SetCellValue("Remark");
            }
            else if (reportType == "24")
            {
                headerRow.CreateCell(0).SetCellValue("Employee Code");
                headerRow.CreateCell(1).SetCellValue("Cleaner Name");
                headerRow.CreateCell(2).SetCellValue("Branch ID");
                headerRow.CreateCell(3).SetCellValue("Branch Name");
                headerRow.CreateCell(4).SetCellValue("Mobile");
                headerRow.CreateCell(5).SetCellValue("Aadhaar");
                headerRow.CreateCell(6).SetCellValue("Status");
                headerRow.CreateCell(7).SetCellValue("Join Date");
            }

            // Apply Header Style
            for (int i = 0; i < 8; i++)
            {
                if (headerRow.GetCell(i) != null)
                {
                    headerRow.GetCell(i).CellStyle = headerStyle;
                }
            }

            // Data Rows
            int rowNumber = 1;

            foreach (JObject item in data)
            {
                IRow row = sheet.CreateRow(rowNumber++);

                if (reportType == "21")
                {
                    row.CreateCell(0).SetCellValue(item["EMP_CODE"]?.ToString() ?? "");
                    row.CreateCell(1).SetCellValue(item["EMP_NAME"]?.ToString() ?? "");
                    row.CreateCell(2).SetCellValue(item["BRANCH_ID"]?.ToString() ?? "");
                    row.CreateCell(3).SetCellValue(item["BRANCH_NAME"]?.ToString() ?? "");
                    row.CreateCell(4).SetCellValue(item["PUNCH_DATE"]?.ToString() ?? "");
                    row.CreateCell(5).SetCellValue(item["PUNCH_IN"]?.ToString() ?? "");
                    row.CreateCell(6).SetCellValue(item["PUNCH_OUT"]?.ToString() ?? "");
                    row.CreateCell(7).SetCellValue(item["REMARK"]?.ToString() ?? "");
                }
                else if (reportType == "24")
                {
                    row.CreateCell(0).SetCellValue(item["EMP_CODE"]?.ToString() ?? "");
                    row.CreateCell(1).SetCellValue(item["EMP_NAME"]?.ToString() ?? "");
                    row.CreateCell(2).SetCellValue(item["BRANCH_ID"]?.ToString() ?? "");
                    row.CreateCell(3).SetCellValue(item["BRANCH_NAME"]?.ToString() ?? "");
                    row.CreateCell(4).SetCellValue(item["MOBILE_NO"]?.ToString() ?? "");
                    row.CreateCell(5).SetCellValue(item["AADHAAR_NO"]?.ToString() ?? "");
                    row.CreateCell(6).SetCellValue(item["STATUS"]?.ToString() ?? "");
                    row.CreateCell(7).SetCellValue(item["JOIN_DATE"]?.ToString() ?? "");
                }
            }

            // Auto Size Columns
            for (int i = 0; i < 8; i++)
            {
                sheet.AutoSizeColumn(i);
            }

            using (var stream = new MemoryStream())
            {
                workbook.Write(stream);

                string fileName =
                    reportType == "21"
                    ? "Cleaner_Punching_Report.xlsx"
                    : "Cleaner_Master_Report.xlsx";

                workbook.Close();

                return File(
                    stream.ToArray(),
                    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
                    fileName
                );
            }
        }

    }

        public async Task<dynamic> UploadDocument(string query, string code, string ApiPath,string token)
        {

            if (IsTokenExpired(token))
            {
                throw new Exception("Token has expired.Please log in again.");
            }

            string data = "";
            using (HttpClient client = new HttpClient())
            {

                string content = JsonConvert.SerializeObject(new { p_query = query, docData = code });
                var buffer = Encoding.UTF8.GetBytes(content);
                var byteContent = new ByteArrayContent(buffer);
                byteContent.Headers.ContentType = new System.Net.Http.Headers.MediaTypeHeaderValue("application/json");

                if (token != null)
                {
                    client.DefaultRequestHeaders.Add("Authorization", $"Bearer {token}");
                }


                string url2 = ApiPath;
                HttpResponseMessage response2 = await client.PostAsync(url2, byteContent);
                if (response2.IsSuccessStatusCode)
                {
                    data = response2.Content.ReadAsStringAsync().Result;
                    data = data.Replace("\"\"", "");
                    data = data.Replace("\'", "");
                    // data = data.Replace(@"""RESP"":", @"");

                }


            }

            data = RemoveSpecialCharacters(data);

            return data;

        }
