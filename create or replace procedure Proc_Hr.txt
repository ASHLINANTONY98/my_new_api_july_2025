create or replace procedure Proc_Hrm_Asset_And_Liability_mvc_new(p_flag         in number,

            (SEQ_ID, DEPENDANT, BANK_NAME, BRANCH, ACCOUNT_TYPE, BALANCE)

            (v_spliter_c(6), v_spliter_c(5), v_spliter_d(1), v_spliter_d(2), v_spliter_d(3),

          insert into TBL_HRM_ASSET_VDA
            (SEQ_ID, DEPENDANT, ASSET_TYPE, QUANTITY, APPROX_VALUE)
          values
            (v_spliter_c(6), v_spliter_c(5), v_spliter_d(1), v_spliter_d(2),

        end if;
      end loop;
    end if;

      update TBL_HRM_ASSET_AND_LIABILITY_DETAILS z

           cash_and_bankbalance,
           personal_loan,
           remark,
           personal_rmrk
    from TBL_HRM_MOVABLE_FORMS
    where seq_id in
    (
        select sequence
        from TBL_HRM_ASSET_AND_LIABILITY_DETAILS
        where seq_id = v_spliter_a(1)
    );
  elsif p_flag = 34 then
    open p_as_outresult for
      select dependant, investment, amount, remark, issuer, units_qty
        from TBL_HRM_ASSET_INVESTMENT
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  elsif p_flag = 35 then
    open p_as_outresult for
      select dependant, other_assets, value, remark, source_of_funds
        from TBL_HRM_OTHER_ASSETS
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  elsif p_flag = 36 then
    open p_as_outresult for
      select f.dependant,
             decode(f.immovable_form, 0, 'Yes', 1, 'No') immovable_form,
             f.property_type, f.details, s.state_name, d.district_name, f.division, f.thaluk, f.village,
             f.distinctive_number, f.total_area, f.type_of_land, f.how_acquired,
             f.name_of_person_from_whom_acq, f.address, f.connection_to_emp, f.cost_of_acq,
             f.present_value, f.annual_income, f.remark,
             f.source_of_funds
        from TBL_HRM_IMMOVABLE_FORM f, STATE_MASTER s, DISTRICT_MASTER d
       where f.state_id = s.state_id(+)
         and f.district_id = d.district_id(+)
         and f.seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  elsif p_flag = 37 then
    open p_as_outresult for
      select dependant, decode(dept_liability, 0, 'Yes', 1, 'No') dept_liability, name, address, nature, amount, remark,
             loan_source, loan_type, sanctioned_amount, outstanding_amount, emi_amount, relationship, purpose
        from TBL_HRM_DEBT_LIABILITY
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  elsif p_flag = 38 then
    open p_as_outresult for
      select dependant, vehicles, registration_number, year_of_purchase, amount_paid, remark,
             payment_mode, lender_name, loan_amount, outstanding_amount, emi_amount, co_applicant
        from TBL_HRM_ASSET_VEHICLES
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));



  -- 39 : header of the PDF (name, code, designation, joining date + the new declaration columns)
  elsif p_flag = 39 then
    open p_as_outresult for
      select e.emp_name,
             e.emp_code,
             dm.designation,
             dd.dep_name     as department,      
             bt.BRANCH_NAME   as branch,     
             e.join_dt,
             p.pan_no,      
             h.declaration_date,
             h.guarantee_details,
             h.integrity_flag,
             h.integrity_details,
             h.consent_flag,
             h.place
        from TBL_HRM_ASSET_AND_LIABILITY h,
             mana0809.employee_master e,
             mana0809.designation_master dm,
             mana0809.department_mst dd,
             mana0809.branch_dtl_new bt,
             mana0809.TBL_EMP_AADHAR_PAN p
       where h.seq_id = v_spliter_a(1)
         and e.emp_code = h.emp_code
         and bt.BRANCH_ID(+) = e.branch_id
         and dd.dep_id(+) = e.department_id
         and p.emp_code(+) = e.emp_code
         and dm.designation_id(+) = e.designation_id;

  -- 40 : bank / deposit accounts (Q2)
  elsif p_flag = 40 then
    open p_as_outresult for
      select dependant, bank_name, branch, account_type, balance
        from TBL_HRM_ASSET_BANK_DEPOSIT
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  -- 41 : insurance / PPF (Q6)
  elsif p_flag = 41 then
    open p_as_outresult for
      select dependant, instrument_type, provider, policy_no, sum_assured
        from TBL_HRM_ASSET_INSURANCE_PPF
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  -- 42 : virtual digital assets (Q8)
  elsif p_flag = 42 then
    open p_as_outresult for
      select dependant, asset_type, quantity, approx_value
        from TBL_HRM_ASSET_VDA
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  -- 43 : recent acquisitions (Q9)
  elsif p_flag = 43 then
    open p_as_outresult for
      select dependant, description, purchase_date, value, source_of_funds, financing_details
        from TBL_HRM_ASSET_RECENT_ACQ
       where seq_id in (select sequence from TBL_HRM_ASSET_AND_LIABILITY_DETAILS where seq_id = v_spliter_a(1));

  -- 44 : annual income (Q10)  - saved once per declaration, so SEQ_ID is the header
  elsif p_flag = 44 then
    open p_as_outresult for
      select fin_year, gross_salary, other_income
        from TBL_HRM_ASSET_INCOME_ANNUAL
       where seq_id = v_spliter_a(1)
       order by fin_year desc;

  -- 45 : income other than salary (Q11)
  elsif p_flag = 45 then
    open p_as_outresult for
      select dependant, nature, source_payer, amount, fin_year
        from TBL_HRM_ASSET_INCOME_OTHER
       where seq_id = v_spliter_a(1);

  -- 46 : other accounts (Q15)
  elsif p_flag = 46 then
    open p_as_outresult for
      select dependant, institution, holding_type, approx_value
        from TBL_HRM_ASSET_OTHER_ACCOUNTS
       where seq_id = v_spliter_a(1);

  -- 47 : declaration status for an employee   0 = still to do, 1 = submitted, 9 = no declaration started
  elsif p_flag = 47 then
    select count(*) into v_ecnt from TBL_HRM_ASSET_AND_LIABILITY where emp_code = v_spliter_a(1);

    if v_ecnt = 0 then
      open p_as_outresult for select '9' status from dual;
    else
      select count(*)
        into V_CNT
        from (select declaration_date
                from TBL_HRM_ASSET_AND_LIABILITY
               where emp_code = v_spliter_a(1)
               order by enter_date desc)
       where rownum = 1
         and declaration_date is not null;

      open p_as_outresult for select to_char(V_CNT) status from dual;
    end if;

  -- 48 : save the final declaration
  --   p_indata = empcode # annual # other # guarantee # integrityFlag # integrityDetails # consent # place # accounts
  --   annual   = fy*gross*other!fy*gross*other!      other = nature*payer*amount*fy!      accounts = institution*type*heldBy*value!
  --   '-' means nothing.  Returns 111 = saved, 222 = already submitted, 333 = no declaration started
  elsif p_flag = 48 then
    v_spliter_b := splitstr(p_indata, '#');

    select count(*) into v_ecnt from TBL_HRM_ASSET_AND_LIABILITY where emp_code = v_spliter_b(1);
    if v_ecnt = 0 then
      open p_as_outresult for select '333' from dual;
      return;
    end if;

    select seq_id
      into seq1
      from (select seq_id
              from TBL_HRM_ASSET_AND_LIABILITY
             where emp_code = v_spliter_b(1)
             order by enter_date desc)
     where rownum = 1;

    select count(*)
      into V_CNT
      from TBL_HRM_ASSET_AND_LIABILITY
     where seq_id = seq1
       and declaration_date is not null;

    if V_CNT > 0 then
      open p_as_outresult for select '222' from dual;
      return;
    end if;

    update TBL_HRM_ASSET_AND_LIABILITY
       set declaration_date  = sysdate,
           guarantee_details = nullif(trim(v_spliter_b(4)), '-'),
           integrity_flag    = to_number(v_spliter_b(5)),
           integrity_details = nullif(trim(v_spliter_b(6)), '-'),
           consent_flag      = to_number(v_spliter_b(7)),
           place             = nullif(trim(v_spliter_b(8)), '-')
     where seq_id = seq1;

    -- Q10 annual income
    if v_spliter_b(2) <> '-' then
      v_spliter_b1 := splitstr(v_spliter_b(2), '!');
      for k in 1 .. v_spliter_b1.count loop
        if (v_spliter_b1(k) is not null) then
          v_spliter_c := splitstr(v_spliter_b1(k), '*');
          insert into TBL_HRM_ASSET_INCOME_ANNUAL (SEQ_ID, FIN_YEAR, GROSS_SALARY, OTHER_INCOME)
          values (seq1, v_spliter_c(1), to_number(v_spliter_c(2)), to_number(nullif(trim(v_spliter_c(3)), '-')));
        end if;
      end loop;
    end if;

    -- Q11 other income
    if v_spliter_b(3) <> '-' then
      v_spliter_b1 := splitstr(v_spliter_b(3), '!');
      for k in 1 .. v_spliter_b1.count loop
        if (v_spliter_b1(k) is not null) then
          v_spliter_c := splitstr(v_spliter_b1(k), '*');
          insert into TBL_HRM_ASSET_INCOME_OTHER (SEQ_ID, NATURE, SOURCE_PAYER, AMOUNT, FIN_YEAR)
          values (seq1, v_spliter_c(1), v_spliter_c(2), to_number(v_spliter_c(3)), v_spliter_c(4));
        end if;
      end loop;
    end if;

    -- Q15 other accounts
    if v_spliter_b(9) <> '-' then
      v_spliter_b1 := splitstr(v_spliter_b(9), '!');
      for k in 1 .. v_spliter_b1.count loop
        if (v_spliter_b1(k) is not null) then
          v_spliter_c := splitstr(v_spliter_b1(k), '*');
          insert into TBL_HRM_ASSET_OTHER_ACCOUNTS (SEQ_ID, DEPENDANT, INSTITUTION, HOLDING_TYPE, APPROX_VALUE)
          values (seq1, v_spliter_c(3), v_spliter_c(1), v_spliter_c(2), to_number(v_spliter_c(4)));
        end if;
      end loop;
    end if;

    commit;
    open p_as_outresult for select '111' from dual;
    return;


  -- 49 : register the PDF in DMS and return the SQL that stores the file   (replaces the old  IF p_flag = 286)
  --   p_indata = seqId ~ empCode ~ fileName     (seqId = header SEQ_ID, flag 29)
  --   the controller then posts this SQL + the PDF to HRMSDocumentUpload.  No quotes in the answer on purpose:
  --   PostInternalPageData removes single quotes from responses.
  --   MANA0809 needs INSERT / UPDATE rights on DMS.TBL_HRM_ASSET_LIABILITY_PDF.
  elsif p_flag = 49 then
    declare
      v_fy_year varchar2(20);
      v_seq     number := to_number(v_spliter_a(1));   -- numeric only, so nothing else can reach the SQL below
    begin
      select to_char(fiscal_year, 'YYYY') || '-' || substr(to_char(add_months(fiscal_year, 12), 'YYYY'), 3)
        into v_fy_year
        from mana0809.fiscal_year_vw;

      select count(*)
        into V_CNT
        from DMS.TBL_HRM_ASSET_LIABILITY_PDF
       where seq_id = v_seq;

      if V_CNT = 0 then
        insert into DMS.TBL_HRM_ASSET_LIABILITY_PDF
          (SEQ_ID, EMP_CODE, DOC_YEAR, FILE_NAME, GENERATED_DATE, GENERATED_BY, STATUS)
        values
          (v_seq, v_spliter_a(2), v_fy_year, v_spliter_a(3), sysdate, v_spliter_a(2), 1);
      else
        update DMS.TBL_HRM_ASSET_LIABILITY_PDF
           set file_name = v_spliter_a(3),
               generated_date = sysdate
         where seq_id = v_seq;
      end if;

      commit;

      open p_as_outresult for
        select 'update DMS.TBL_HRM_ASSET_LIABILITY_PDF set PDF_DOC = :DOCUMENT where SEQ_ID = ' || v_seq as RES
          from dual;
    end;

  -- 50 : save an Annexure-II (Conflict of Interest) disclosure - the yearly one, or a new one raised during the year
  --   p_indata = empcode # q1flag # q1rows # q2flag # q2rows # q3flag # q3text # place
  --   q1rows = entity*relationship*relative*nature*value!      q2rows = activity*entity*relative*competes!
  --   '-' means nothing.  Returns 111 = saved, 333 = no declaration started for this employee
  elsif p_flag = 50 then
    v_spliter_b := splitstr(p_indata, '#');

    select count(*) into v_ecnt from TBL_HRM_ASSET_AND_LIABILITY where emp_code = v_spliter_b(1);
    if v_ecnt = 0 then
      open p_as_outresult for select '333' from dual;
      return;
    end if;

    select seq_id
      into seq1
      from (select seq_id
              from TBL_HRM_ASSET_AND_LIABILITY
             where emp_code = v_spliter_b(1)
             order by enter_date desc)
     where rownum = 1;

    declare
      v_coi number;
    begin
      v_coi := SEQ_HRM_COI.NEXTVAL;

      insert into TBL_HRM_ASSET_COI
        (COI_ID, SEQ_ID, EMP_CODE, DISCLOSURE_DATE, Q1_FLAG, Q2_FLAG, Q3_FLAG, Q3_DESCRIPTION, PLACE)
      values
        (v_coi, seq1, v_spliter_b(1), sysdate,
         to_number(v_spliter_b(2)), to_number(v_spliter_b(4)), to_number(v_spliter_b(6)),
         nullif(trim(v_spliter_b(7)), '-'), nullif(trim(v_spliter_b(8)), '-'));

      -- Q1 interests
      if v_spliter_b(3) <> '-' then
        v_spliter_b1 := splitstr(v_spliter_b(3), '!');
        for k in 1 .. v_spliter_b1.count loop
          if (v_spliter_b1(k) is not null) then
            v_spliter_c := splitstr(v_spliter_b1(k), '*');
            insert into TBL_HRM_ASSET_COI_INTEREST
              (COI_ID, ENTITY_NAME, ENTITY_RELATION, RELATIVE_NAME_REL, NATURE_OF_INTEREST, APPROX_VALUE)
            values
              (v_coi, v_spliter_c(1), v_spliter_c(2), nullif(trim(v_spliter_c(3)), '-'), v_spliter_c(4), v_spliter_c(5));
          end if;
        end loop;
      end if;

      -- Q2 outside activities
      if v_spliter_b(5) <> '-' then
        v_spliter_b1 := splitstr(v_spliter_b(5), '!');
        for k in 1 .. v_spliter_b1.count loop
          if (v_spliter_b1(k) is not null) then
            v_spliter_c := splitstr(v_spliter_b1(k), '*');
            insert into TBL_HRM_ASSET_COI_ACTIVITY
              (COI_ID, NATURE_OF_ACTIVITY, ENTITY_NAME, RELATIVE_NAME_REL, COMPETES)
            values
              (v_coi, v_spliter_c(1), v_spliter_c(2), nullif(trim(v_spliter_c(3)), '-'), v_spliter_c(4));
          end if;
        end loop;
      end if;

      commit;
    end;

    open p_as_outresult for select '111' from dual;
    return;

  -- 51 - 54 : Annexure-II for the PDF (latest disclosure of the declaration; p_indata = header SEQ_ID)
  elsif p_flag = 51 then
    open p_as_outresult for
      select coi_id, disclosure_date, q1_flag, q2_flag, q3_flag, q3_description, place, reviewed_by, outcome
        from TBL_HRM_ASSET_COI
       where coi_id = (select max(coi_id) from TBL_HRM_ASSET_COI where seq_id = v_spliter_a(1));

  elsif p_flag = 52 then
    open p_as_outresult for
      select entity_name, entity_relation, relative_name_rel, nature_of_interest, approx_value
        from TBL_HRM_ASSET_COI_INTEREST
       where coi_id = (select max(coi_id) from TBL_HRM_ASSET_COI where seq_id = v_spliter_a(1));

  elsif p_flag = 53 then
    open p_as_outresult for
      select nature_of_activity, entity_name, relative_name_rel, competes
        from TBL_HRM_ASSET_COI_ACTIVITY
       where coi_id = (select max(coi_id) from TBL_HRM_ASSET_COI where seq_id = v_spliter_a(1));

  elsif p_flag = 54 then
    open p_as_outresult for
      select conflict_identified, mitigation_measure, approved_by, approved_date
        from TBL_HRM_ASSET_COI_MITIGATION
       where coi_id = (select max(coi_id) from TBL_HRM_ASSET_COI where seq_id = v_spliter_a(1));

  ----------------------------------------------------------------------
 
  end if;
exception
  when others then
    -- p_errorMsg := SUBSTR(SQLERRM, 1, 150) || '- Error Code :-' || SQLCODE;
    -- rollback;

    rollback;

    v_ErrID    := seq_hrms_errorid.nextval;
    p_errormsg := 'Oops..!Something Went Wrong...!' || v_ErrID;
    v_ErrDesc  := SUBSTR(SQLERRM, 1, 150); -- || v_spliter_a(2);
    v_ErrDtl   := v_ErrDesc || dbms_utility.format_error_backtrace ||
                  dbms_utility.format_error_stack ||
                  dbms_utility.format_call_stack;
    insert into tbl_hrms_error_dtl
      (error_id, proc_name, error_description, error_details, tra_dt)
    values
      (v_ErrID,
       'Proc_Hrm_Asset_And_Liability_mvc_new',
       v_ErrDesc,
       v_ErrDtl,
       sysdate);
    commit;
end Proc_Hrm_Asset_And_Liability_mvc_new;
